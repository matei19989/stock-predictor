## Task 26: Dockerfile + docker-compose

**Files:**
- Create: `src/backend/Dockerfile`
- Modify: `docker-compose.yml` (backend service block only)
- Create: `.env` (repo root)

The Dockerfile uses a multi-stage build: `sdk:9.0` for restore + publish, `aspnet:9.0` for the runtime image. The build context is `src/backend/` (not the repo root), so all `COPY` paths are relative to that directory. Project files are copied first (for layer caching on `dotnet restore`), then the full source, then `dotnet publish` produces a self-contained output. The `docker-compose.yml` backend service passes environment variables for connection string, JWT, and ML service URL. The JWT secret is read from a `.env` file (already in `.gitignore`).

- [ ] **Step 1: Create Dockerfile**

`src/backend/Dockerfile`
```dockerfile
FROM mcr.microsoft.com/dotnet/aspnet:9.0 AS base
WORKDIR /app
EXPOSE 5000

FROM mcr.microsoft.com/dotnet/sdk:9.0 AS build
WORKDIR /src
COPY ["Domain/StockPredictor.Domain.csproj", "Domain/"]
COPY ["Application/StockPredictor.Application.csproj", "Application/"]
COPY ["Infrastructure/StockPredictor.Infrastructure.csproj", "Infrastructure/"]
COPY ["API/StockPredictor.API.csproj", "API/"]
RUN dotnet restore "API/StockPredictor.API.csproj"
COPY . .
RUN dotnet publish "API/StockPredictor.API.csproj" -c Release -o /app/publish --no-restore

FROM base AS final
WORKDIR /app
COPY --from=build /app/publish .
ENTRYPOINT ["dotnet", "StockPredictor.API.dll"]
```

## Task 28: AuthServiceTests

**Files:**
- Create: `tests/backend/StockPredictor.Tests.Unit/Services/AuthServiceTests.cs`

Six tests covering the core auth flows:
- `RegisterAsync` — happy path returns JWT token with correct username/email
- `RegisterAsync` — duplicate email throws `ConflictException` with "email" in message
- `RegisterAsync` — duplicate username throws `ConflictException` with "username" in message
- `LoginAsync` — wrong password throws `UnauthorizedException`
- `LoginAsync` — email not found throws `UnauthorizedException`
- `LoginAsync` — valid credentials returns JWT token

The `AuthService` constructor takes: `IUserRepository`, `IWatchlistService`, `IConfiguration`, `ILogger<AuthService>`. Configuration is built in-memory with JWT settings. Logger uses `NullLogger`. `RegisterAsync` calls `EmailExistsAsync` then `UsernameExistsAsync` then `AddAsync` then `SeedDefaultsAsync`. Note: `RegisterAsync` lower-cases email before checking (`request.Email.ToLower()`), so the mock setup for `EmailExistsAsync` must match the lower-cased value.

- [ ] **Step 1: Create AuthServiceTests**

```csharp
using FluentAssertions;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.Logging.Abstractions;
using Moq;
using StockPredictor.Application.DTOs.Auth;
using StockPredictor.Application.Exceptions;
using StockPredictor.Application.Interfaces.Repositories;
using StockPredictor.Application.Interfaces.Services;
using StockPredictor.Domain.Entities;
using StockPredictor.Infrastructure.Services;

namespace StockPredictor.Tests.Unit.Services;

public class AuthServiceTests
{
    private readonly Mock<IUserRepository> _users = new();
    private readonly Mock<IWatchlistService> _watchlist = new();
    private readonly IConfiguration _config;
    private readonly AuthService _sut;

    public AuthServiceTests()
    {
        _config = new ConfigurationBuilder()
            .AddInMemoryCollection(new Dictionary<string, string?>
            {
                ["Jwt:Key"] = "test-secret-key-minimum-32-characters-long-xxx",
                ["Jwt:Issuer"] = "StockPredictor",
                ["Jwt:Audience"] = "StockPredictor",
                ["Jwt:ExpiryDays"] = "7"
            })
            .Build();

        _sut = new AuthService(_users.Object, _watchlist.Object, _config,
            NullLogger<AuthService>.Instance);
    }

    [Fact]
    public async Task RegisterAsync_ValidData_ReturnsTokenWithCorrectUsername()
    {
        _users.Setup(r => r.EmailExistsAsync("test@example.com")).ReturnsAsync(false);
        _users.Setup(r => r.UsernameExistsAsync("alice")).ReturnsAsync(false);
        _users.Setup(r => r.AddAsync(It.IsAny<User>())).Returns(Task.CompletedTask);
        _watchlist.Setup(w => w.SeedDefaultsAsync(It.IsAny<Guid>())).Returns(Task.CompletedTask);

        var result = await _sut.RegisterAsync(new RegisterRequest
        {
            Username = "alice",
            Email = "test@example.com",
            Password = "password123"
        });

        result.Token.Should().NotBeNullOrEmpty();
        result.Username.Should().Be("alice");
        result.Email.Should().Be("test@example.com");
    }

    [Fact]
    public async Task RegisterAsync_DuplicateEmail_ThrowsConflictException()
    {
        _users.Setup(r => r.EmailExistsAsync("taken@example.com")).ReturnsAsync(true);

        await _sut.Invoking(s => s.RegisterAsync(new RegisterRequest
            {
                Username = "alice",
                Email = "taken@example.com",
                Password = "password123"
            }))
            .Should().ThrowAsync<ConflictException>()
            .WithMessage("*email*");
    }

    [Fact]
    public async Task RegisterAsync_DuplicateUsername_ThrowsConflictException()
    {
        _users.Setup(r => r.EmailExistsAsync(It.IsAny<string>())).ReturnsAsync(false);
        _users.Setup(r => r.UsernameExistsAsync("alice")).ReturnsAsync(true);

        await _sut.Invoking(s => s.RegisterAsync(new RegisterRequest
            {
                Username = "alice",
                Email = "new@example.com",
                Password = "password123"
            }))
            .Should().ThrowAsync<ConflictException>()
            .WithMessage("*username*");
    }

    [Fact]
    public async Task LoginAsync_WrongPassword_ThrowsUnauthorizedException()
    {
        var user = new User
        {
            Id = Guid.NewGuid(),
            Username = "alice",
            Email = "test@example.com",
            PasswordHash = BCrypt.Net.BCrypt.HashPassword("correct_password"),
            CreatedAt = DateTime.UtcNow
        };

        _users.Setup(r => r.GetByEmailAsync("test@example.com")).ReturnsAsync(user);

        await _sut.Invoking(s => s.LoginAsync(new LoginRequest
            {
                Email = "test@example.com",
                Password = "wrong_password"
            }))
            .Should().ThrowAsync<UnauthorizedException>();
    }

    [Fact]
    public async Task LoginAsync_EmailNotFound_ThrowsUnauthorizedException()
    {
        _users.Setup(r => r.GetByEmailAsync(It.IsAny<string>())).ReturnsAsync((User?)null);

        await _sut.Invoking(s => s.LoginAsync(new LoginRequest
            {
                Email = "ghost@example.com",
                Password = "password123"
            }))
            .Should().ThrowAsync<UnauthorizedException>();
    }

    [Fact]
    public async Task LoginAsync_ValidCredentials_ReturnsToken()
    {
        var user = new User
        {
            Id = Guid.NewGuid(),
            Username = "alice",
            Email = "test@example.com",
            PasswordHash = BCrypt.Net.BCrypt.HashPassword("correct_password"),
            CreatedAt = DateTime.UtcNow
        };

        _users.Setup(r => r.GetByEmailAsync("test@example.com")).ReturnsAsync(user);

        var result = await _sut.LoginAsync(new LoginRequest
        {
            Email = "test@example.com",
            Password = "correct_password"
        });

        result.Token.Should().NotBeNullOrEmpty();
        result.Username.Should().Be("alice");
    }
}
```

### PredictionServiceTests (4 tests)

The `PredictionService` constructor takes: `IPredictionRepository`, `IStockRepository`, `IMlServiceClient`, `ILogger<PredictionService>`.

- `GetOrCreateAsync` — cache hit: returns cached prediction, never calls ML
- `GetOrCreateAsync` — cache miss: calls ML, persists result, returns with 24h expiry
- `GetOrCreateAsync` — unsupported horizon: ML throws `HorizonNotSupportedException`, propagates without persisting
- `GetOrCreateAsync` — stock not found: throws `NotFoundException`

### WatchlistServiceTests (4 tests)

The `WatchlistService` constructor takes 6 parameters: `IWatchlistRepository`, `IStockRepository`, `IStockPriceRepository`, `IPredictionRepository`, `IStockService`, `ILogger<WatchlistService>`.

- `AddAsync` — already in watchlist: throws `ConflictException` (uses `IStockService.EnsureStockExistsAsync` then `IWatchlistRepository.ExistsAsync`)
- `AddAsync` — new stock: calls `IWatchlistRepository.AddAsync`
- `RemoveAsync` — item not found: throws `NotFoundException` (uses `IStockRepository.GetByTickerAsync` then `IWatchlistRepository.GetItemAsync`)
- `RemoveAsync` — item exists: calls `IWatchlistRepository.RemoveAsync`

**Key difference from AddAsync:** `RemoveAsync` uses `IStockRepository.GetByTickerAsync` (not `IStockService.EnsureStockExistsAsync`), because removing shouldn't trigger an ML fetch for unknown tickers.

```csharp
using FluentAssertions;
using Microsoft.Extensions.Logging.Abstractions;
using Moq;
using StockPredictor.Application.Exceptions;
using StockPredictor.Application.Interfaces.External;
using StockPredictor.Application.Interfaces.Repositories;
using StockPredictor.Domain.Entities;
using StockPredictor.Infrastructure.Services;

namespace StockPredictor.Tests.Unit.Services;

public class PredictionServiceTests
{
    private readonly Mock<IPredictionRepository> _predictions = new();
    private readonly Mock<IStockRepository> _stocks = new();
    private readonly Mock<IMlServiceClient> _ml = new();
    private readonly PredictionService _sut;

    public PredictionServiceTests()
    {
        _sut = new PredictionService(_predictions.Object, _stocks.Object, _ml.Object,
            NullLogger<PredictionService>.Instance);
    }

    [Fact]
    public async Task GetOrCreateAsync_CacheHit_ReturnsCachedAndSkipsMl()
    {
        var stock = new Stock { Id = Guid.NewGuid(), Ticker = "AAPL" };
        var cached = new Prediction
        {
            Id = Guid.NewGuid(),
            StockId = stock.Id,
            Horizon = "3m",
            Signal = "Buy",
            Confidence = 0.7,
            ProbabilitiesJson = """{"Buy":0.7,"Hold":0.3}""",
            FeaturesUsed = 22,
            LowConfidence = false,
            CreatedAt = DateTime.UtcNow.AddHours(-1),
            ExpiresAt = DateTime.UtcNow.AddHours(23)
        };

        _stocks.Setup(r => r.GetByTickerAsync("AAPL")).ReturnsAsync(stock);
        _predictions.Setup(r => r.GetValidAsync(stock.Id, "3m")).ReturnsAsync(cached);

        var result = await _sut.GetOrCreateAsync("AAPL", "3m");

        result.Signal.Should().Be("Buy");
        result.Confidence.Should().BeApproximately(0.7, 0.001);
        _ml.Verify(m => m.PredictAsync(It.IsAny<string>(), It.IsAny<string>()), Times.Never);
    }

    [Fact]
    public async Task GetOrCreateAsync_CacheMiss_CallsMlAndPersists()
    {
        var stock = new Stock { Id = Guid.NewGuid(), Ticker = "AAPL" };
        var mlResponse = new MlPredictResponse(
            "AAPL", "3m", "Buy", 0.65,
            new Dictionary<string, double> { ["Buy"] = 0.65, ["Hold"] = 0.35 },
            22, DateTime.UtcNow.ToString("o"), false);

        _stocks.Setup(r => r.GetByTickerAsync("AAPL")).ReturnsAsync(stock);
        _predictions.Setup(r => r.GetValidAsync(stock.Id, "3m")).ReturnsAsync((Prediction?)null);
        _ml.Setup(m => m.PredictAsync("AAPL", "3m")).ReturnsAsync(mlResponse);
        _predictions.Setup(r => r.AddAsync(It.IsAny<Prediction>())).Returns(Task.CompletedTask);

        var result = await _sut.GetOrCreateAsync("AAPL", "3m");

        result.Signal.Should().Be("Buy");
        result.Confidence.Should().BeApproximately(0.65, 0.001);
        result.ExpiresAt.Should().BeCloseTo(DateTime.UtcNow.AddHours(24), TimeSpan.FromMinutes(1));
        _predictions.Verify(r => r.AddAsync(It.IsAny<Prediction>()), Times.Once);
    }

    [Fact]
    public async Task GetOrCreateAsync_HorizonNotSupported_PropagatesException()
    {
        var stock = new Stock { Id = Guid.NewGuid(), Ticker = "AAPL" };
        _stocks.Setup(r => r.GetByTickerAsync("AAPL")).ReturnsAsync(stock);
        _predictions.Setup(r => r.GetValidAsync(stock.Id, "6m")).ReturnsAsync((Prediction?)null);
        _ml.Setup(m => m.PredictAsync("AAPL", "6m"))
           .ThrowsAsync(new HorizonNotSupportedException("6m"));

        await _sut.Invoking(s => s.GetOrCreateAsync("AAPL", "6m"))
                  .Should().ThrowAsync<HorizonNotSupportedException>();

        _predictions.Verify(r => r.AddAsync(It.IsAny<Prediction>()), Times.Never);
    }

    [Fact]
    public async Task GetOrCreateAsync_StockNotFound_ThrowsNotFoundException()
    {
        _stocks.Setup(r => r.GetByTickerAsync("UNKNOWN")).ReturnsAsync((Stock?)null);

        await _sut.Invoking(s => s.GetOrCreateAsync("UNKNOWN", "3m"))
                  .Should().ThrowAsync<NotFoundException>();
    }
}
```

```csharp
    public async Task AddAsync_WhenAlreadyInWatchlist_ThrowsConflictException()
    {
        var userId = Guid.NewGuid();
        var stock = new Stock { Id = Guid.NewGuid(), Ticker = "AAPL" };

        _stockService.Setup(s => s.EnsureStockExistsAsync("AAPL")).ReturnsAsync(stock);
        _watchlist.Setup(w => w.ExistsAsync(userId, stock.Id)).ReturnsAsync(true);

        await _sut.Invoking(s => s.AddAsync(userId, "AAPL"))
                  .Should().ThrowAsync<ConflictException>()
                  .WithMessage("*already in your watchlist*");
    }

    [Fact]
    public async Task AddAsync_NewStock_CallsRepositoryAdd()
    {
        var userId = Guid.NewGuid();
        var stock = new Stock { Id = Guid.NewGuid(), Ticker = "NVDA" };

        _stockService.Setup(s => s.EnsureStockExistsAsync("NVDA")).ReturnsAsync(stock);
        _watchlist.Setup(w => w.ExistsAsync(userId, stock.Id)).ReturnsAsync(false);
        _watchlist.Setup(w => w.AddAsync(It.IsAny<WatchlistItem>())).Returns(Task.CompletedTask);

        await _sut.AddAsync(userId, "NVDA");

        _watchlist.Verify(w => w.AddAsync(It.Is<WatchlistItem>(
            i => i.UserId == userId && i.StockId == stock.Id)), Times.Once);
    }

    [Fact]
    public async Task RemoveAsync_ItemNotInWatchlist_ThrowsNotFoundException()
    {
        var userId = Guid.NewGuid();
        var stock = new Stock { Id = Guid.NewGuid(), Ticker = "AAPL" };

        _stocks.Setup(s => s.GetByTickerAsync("AAPL")).ReturnsAsync(stock);
        _watchlist.Setup(w => w.GetItemAsync(userId, stock.Id)).ReturnsAsync((WatchlistItem?)null);

        await _sut.Invoking(s => s.RemoveAsync(userId, "AAPL"))
                  .Should().ThrowAsync<NotFoundException>();
    }

    [Fact]
    public async Task RemoveAsync_ItemExists_CallsRepositoryRemove()
    {
        var userId = Guid.NewGuid();
        var stock = new Stock { Id = Guid.NewGuid(), Ticker = "AAPL" };
        var item = new WatchlistItem { Id = Guid.NewGuid(), UserId = userId, StockId = stock.Id };

        _watchlist.Setup(w => w.GetItemAsync(userId, stock.Id)).ReturnsAsync(item);
        _watchlist.Setup(w => w.RemoveAsync(item)).Returns(Task.CompletedTask);
```
